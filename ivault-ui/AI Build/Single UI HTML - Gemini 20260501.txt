// ==========================================
// 1. DATA MODELS & TYPES
// ==========================================
interface Message {
  id: string;
  sender: string;
  initials: string;
  text: string;
  time: string;
  isMe: boolean;
}

interface EcmRecord {
  id: string;
  recordName: string;
  type: 'folder' | 'record';
  contentFile?: { name: string; type: 'pdf' | 'docx' | 'xlsx'; size: string; };
  metadata: Record<string, any>;
  modifiedDate: string;
  parentId?: string | null;
}

interface Workspace {
  id: string;
  name: string;
  members: number;
  description: string;
  status: 'On-Going' | 'Closed';
  cabinetFiles: EcmRecord[];
  discussion: Message[];
}

interface WorkflowTask {
  id: string;
  title: string;
  priority: 'High' | 'Medium' | 'Low';
  status: 'Incoming' | 'In-Progress' | 'Completed';
  initiatedBy: string;
  dueDate: string;
  currentStep: number;
  steps: string[];
  record: EcmRecord;
}

/** DISCUSSION COMPONENT */
@Component({
  selector: 'app-discussion',
  standalone: true,
  imports: [CommonModule],
  template: `
    <div class="flex-1 flex flex-col bg-white dark:bg-slate-800 overflow-hidden h-full">
      <div class="p-4 border-b border-slate-100 dark:border-slate-700 bg-slate-50/50 flex justify-between items-center">
        <div class="flex items-center gap-3">
          <div class="w-2 h-2 bg-green-500 rounded-full animate-pulse"></div>
          <span class="text-[10px] font-black text-slate-400 uppercase tracking-widest">Active Discussion</span>
        </div>
      </div>
      <div class="flex-1 overflow-y-auto p-6 space-y-6">
        @for (msg of messages; track msg.id) {
          <div class="flex items-start gap-4" [class.flex-row-reverse]="msg.isMe">
            <div class="w-8 h-8 rounded-full flex items-center justify-center shrink-0 text-[10px] font-black"
                 [class.bg-indigo-600]="msg.isMe" [class.text-white]="msg.isMe"
                 [class.bg-slate-200]="!msg.isMe" [class.text-slate-600]="!msg.isMe">
              {{ msg.initials }}
            </div>
            <div class="max-w-[70%] space-y-1" [class.text-right]="msg.isMe">
              <div class="flex items-center gap-2" [class.flex-row-reverse]="msg.isMe">
                <span class="text-[10px] font-bold text-slate-900 dark:text-slate-100">{{ msg.sender }}</span>
                <span class="text-[9px] text-slate-400">{{ msg.time }}</span>
              </div>
              <div class="p-3 rounded-2xl text-xs shadow-sm"
                   [class.bg-indigo-600]="msg.isMe" [class.text-white]="msg.isMe" [class.rounded-tr-none]="msg.isMe"
                   [class.bg-slate-100]="!msg.isMe" [class.text-slate-700]="!msg.isMe" [class.rounded-tl-none]="!msg.isMe">
                {{ msg.text }}
              </div>
            </div>
          </div>
        }
      </div>
      <div class="p-4 border-t border-slate-100 dark:border-slate-700">
        <div class="flex items-center gap-2 bg-slate-50 dark:bg-slate-900 border border-slate-200 rounded-xl px-4 py-2">
          <input #msgInput type="text" (keyup.enter)="onSend(msgInput)" placeholder="Type a message..." class="flex-1 bg-transparent outline-none text-xs py-1">
          <button (click)="onSend(msgInput)" class="p-2 text-indigo-600 hover:bg-indigo-50 rounded-lg">
            <svg xmlns="http://www.w3.org/2000/svg" width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5"><path d="m22 2-7 20-4-9-9-4Z"/><path d="M22 2 11 13"/></svg>
          </button>
        </div>
      </div>
    </div>
  `
})
class DiscussionComponent {
  @Input() messages: Message[] = [];
  @Output() sendMessage = new EventEmitter<string>();
  onSend(input: HTMLInputElement) {
    if (input.value.trim()) {
      this.sendMessage.emit(input.value);
      input.value = '';
    }
  }
}

export class App {
  isDarkMode = signal(false);
  tabs = ['Dashboard', 'Repository', 'Workflows', 'News'];
  activeTab = signal('Dashboard');
  currentView = signal<'discussion' | 'cabinet'>('discussion');
  
  viewingWorkspaceId = signal<string | null>(null);
  selectedFolderId = signal<string | null>('root1');
  selectedRecordId = signal<string | null>(null);
  
  // Workflow state
  workflowFilter = signal<'Incoming' | 'In-Progress'>('Incoming');
  selectedWorkflowId = signal<string | null>(null);

  workspaces: Workspace[] = [
    { 
      id: 'ws1', 
      name: 'Project Alpha Collaboration', 
      members: 4, 
      description: 'Phase 2 alpha rollout assets.', 
      status: 'On-Going',
      cabinetFiles: [],
      discussion: []
    }
  ];

  // Computed
  filteredWorkflows = computed(() => MOCK_WORKFLOWS.filter(w => w.status === this.workflowFilter()));
  selectedWorkflow = computed(() => MOCK_WORKFLOWS.find(w => w.id === this.selectedWorkflowId()));
  activeWorkspace = computed(() => this.workspaces.find(w => w.id === this.viewingWorkspaceId()));
  catalogRoots = computed(() => MOCK_RECORDS.filter(r => r.type === 'folder' && r.parentId === null));

  // Handlers
  onNewMessage(text: string) {
    const ws = this.activeWorkspace();
    if (ws) {
      ws.discussion.push({
        id: Date.now().toString(), sender: 'John Doe', initials: 'JD', text: text,
        time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }), isMe: true
      });
    }
  }

  selectFolder(id: string) { this.selectedFolderId.set(id); }
  getKeys(obj: any) { return obj ? Object.keys(obj) : []; }
}
